cmake_minimum_required(VERSION 3.16)
project(psbbv_applications_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(psbbv_app_catalogue src/catalogue.c)
target_include_directories(psbbv_app_catalogue PUBLIC include)
target_compile_options(psbbv_app_catalogue PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(test_app_catalogue tests/test_app_catalogue.c)
target_link_libraries(test_app_catalogue PRIVATE psbbv_app_catalogue)
target_compile_options(test_app_catalogue PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME app_catalogue COMMAND test_app_catalogue)
